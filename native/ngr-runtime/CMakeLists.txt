cmake_minimum_required(VERSION 3.22.1)
project(ngr_runtime LANGUAGES C CXX)
add_library(ngr_runtime SHARED src/jni_bridge.cpp src/runtime_info.cpp src/vulkan_probe.cpp src/capability_registry.cpp src/backend_abi_validation.cpp src/backend_registry.cpp src/backend_abi_c_check.c)
target_include_directories(ngr_runtime PRIVATE include)
target_compile_features(ngr_runtime PRIVATE cxx_std_20)
target_compile_options(ngr_runtime PRIVATE -Wall -Wextra -Wpedantic)
find_library(android-lib android)
find_library(log-lib log)
find_library(vulkan-lib vulkan)
target_link_libraries(ngr_runtime PRIVATE ${android-lib} ${log-lib} ${vulkan-lib} N64Recomp N64RecompElf)
